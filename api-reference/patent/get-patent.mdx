import ViewParam from "/snippets/view-param.mdx";
import EntityMeta from "/snippets/entity-meta.mdx";

## Parámetros de ruta

<ParamField path="patent_id" type="string" required>
  Identificador de la patente. Ejemplo: `6ab847915418a51e32855aa6`.
</ParamField>

## Parámetros de consulta

<ViewParam />

## Respuesta

<ResponseField name="data" type="object">
  La patente solicitada.
  <Expandable title="propiedades">
    <ResponseField name="id" type="string">
      Identificador de la patente.
    </ResponseField>
    <ResponseField name="title" type="string">
      Título principal.
    </ResponseField>
    <ResponseField name="titles" type="object[]">
      Títulos por fuente (`title`, `lang`, `source`, `provenance`).
    </ResponseField>
    <ResponseField name="language" type="string | null">
      Idioma (código ISO 639-1).
    </ResponseField>
    <ResponseField name="product_types" type="object[]">
      Tipos de producto por fuente (`name`, `source`). Por ejemplo:
      `"Patente de invención"` según Minciencias.
    </ResponseField>
    <ResponseField name="types" type="object[]">
      Tipos por fuente con `type`, `level`, `source` y `provenance`.
    </ResponseField>
    <ResponseField name="ranking" type="object[]">
      Clasificaciones de la patente (`rank`, `source`, `date`).
    </ResponseField>
    <ResponseField name="authors" type="object[]">
      Autores con sus afiliaciones e identificadores externos.
    </ResponseField>
    <ResponseField name="authors_count" type="integer">
      Cantidad de autores.
    </ResponseField>
    <ResponseField name="groups" type="object[]">
      Grupos de investigación asociados (`id`, `name`).
    </ResponseField>
    <ResponseField name="external_ids" type="object[]">
      Identificadores externos (`id`, `source`, `provenance`).
    </ResponseField>
    <ResponseField name="external_urls" type="object[]">
      Enlaces externos (`url`, `source`).
    </ResponseField>
    <ResponseField name="updated" type="object[]">
      Última actualización por fuente (`source`, `time` en segundos Unix).
    </ResponseField>
  </Expandable>
</ResponseField>
