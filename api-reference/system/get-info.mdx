## Respuesta

<ResponseField name="db_update" type="integer">
  Fecha de la última actualización de la base de datos en segundos Unix.
</ResponseField>

<ResponseField name="total_products" type="integer">
  Total de productos bibliográficos.
</ResponseField>

<ResponseField name="total_authors" type="integer">
  Total de autores.
</ResponseField>

<ResponseField name="total_institutions" type="integer">
  Total de instituciones.
</ResponseField>

<ResponseField name="total_faculties" type="integer">
  Total de facultades.
</ResponseField>

<ResponseField name="total_departments" type="integer">
  Total de departamentos académicos.
</ResponseField>

<ResponseField name="total_groups" type="integer">
  Total de grupos de investigación.
</ResponseField>

<ResponseField name="total_patents" type="integer">
  Total de patentes.
</ResponseField>

<ResponseField name="total_projects" type="integer">
  Total de proyectos.
</ResponseField>

<ResponseField name="total_sources" type="integer">
  Total de fuentes (revistas y otros tipos de publicación).
</ResponseField>

<ResponseField name="total_news" type="integer">
  Total de noticias.
</ResponseField>

</ResponseField>
