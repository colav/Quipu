---

import ViewParam from "/snippets/view-param.mdx";
import EntityMeta from "/snippets/entity-meta.mdx";

## Parámetros de ruta

<ParamField path="person_id" type="string" required>
  Identificador del autor. Ejemplo: `0000000001`.
</ParamField>

## Parámetros de consulta

<ViewParam />

## Respuesta

<ResponseField name="data" type="object">
  Perfil del autor.
  <Expandable title="propiedades">
    <ResponseField name="id" type="string">
      Identificador del autor.
    </ResponseField>
    <ResponseField name="full_name" type="string | null">
      Nombre completo.
    </ResponseField>
    <ResponseField name="first_names" type="string[]">
      Nombres.
    </ResponseField>
    <ResponseField name="last_names" type="string[]">
      Apellidos.
    </ResponseField>
    <ResponseField name="initials" type="string | null">
      Iniciales de los nombres.
    </ResponseField>
    <ResponseField name="aliases" type="string[]">
      Variantes del nombre con las que aparece en las fuentes.
    </ResponseField>
    <ResponseField name="sex" type="string | null">
      Sexo.
    </ResponseField>
    <ResponseField name="affiliations" type="object[]">
      Afiliaciones del autor (instituciones, facultades, departamentos y grupos).
      <Expandable title="propiedades de cada afiliación">
        <ResponseField name="id" type="string | null">
          Identificador de la afiliación.
        </ResponseField>
        <ResponseField name="name" type="string | null">
          Nombre de la afiliación.
        </ResponseField>
        <ResponseField name="types" type="object[]">
          Tipos de la afiliación por fuente (`type`, `source`, `provenance`).
        </ResponseField>
        <ResponseField name="start_date" type="integer | string | null">
          Fecha de inicio de la vinculación en segundos Unix. `-1` si no tiene fin registrado.
        </ResponseField>
        <ResponseField name="end_date" type="integer | string | null">
          Fecha de fin de la vinculación en segundos Unix. `-1` si sigue activo.
        </ResponseField>
        <ResponseField name="years" type="integer[]">
          Años con producción registrada en esta afiliación.
        </ResponseField>
        <ResponseField name="position" type="string | null">
          Cargo o posición en la afiliación.
        </ResponseField>
        <ResponseField name="external_urls" type="object[]">
          URLs externas de la afiliación.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="external_ids" type="object[]">
      Identificadores externos (Scienti, ORCID, Scholar, ResearchGate, OpenAlex, entre otros).
      Campos: `id`, `source`, `provenance`.
    </ResponseField>
    <ResponseField name="citations_count" type="object[]">
      Citas por fuente (`count`, `source`).
    </ResponseField>
    <ResponseField name="products_count" type="integer | null">
      Cantidad total de productos de investigación.
    </ResponseField>
    <ResponseField name="h_index" type="integer | null">
      Índice h.
    </ResponseField>
    <ResponseField name="h5_index" type="integer | null">
      Índice h de los últimos 5 años.
    </ResponseField>
    <ResponseField name="degrees" type="object[]">
      Formación académica (`degree`, `date`, `source`, `institutions`).
    </ResponseField>
    <ResponseField name="ranking" type="object[]">
      Categorías y clasificaciones: tipo de contrato, jornada, categoría laboral
      y Minciencias (`rank`, `source`, `date`).
    </ResponseField>
    <ResponseField name="subjects" type="object[]">
      Áreas de conocimiento agrupadas por fuente (por ejemplo, OECD).
    </ResponseField>
    <ResponseField name="keywords" type="string[]">
      Palabras clave.
    </ResponseField>
    <ResponseField name="related_works" type="array">
      Trabajos relacionados.
    </ResponseField>
    <ResponseField name="updated" type="object[]">
      Última actualización por fuente (`source`, `time` en segundos Unix).
    </ResponseField>
  </Expandable>
</ResponseField>

<EntityMeta />

<Note>
  La API no publica documentos de identidad, fecha o lugar de nacimiento, edad
  ni estado civil. De `external_ids` solo se publican las fuentes `linkedin`,
  `openalex`, `orcid`, `publons`, `researchgate`, `scholar`, `scienti`,
  `scopus`, `ssrn` y `wos`.
</Note>
