title: "Proyectos de un autor"
api: "GET https://api.dev.impactu.colav.co/person/{person_id}/research/projects"
description: "Lista de forma paginada los proyectos de investigación de un autor."
---

import PaginationParams from "/snippets/pagination-params.mdx";
import MetaResponse from "/snippets/meta-response.mdx";

## Parámetros de ruta

<ParamField path="person_id" type="string" required>
  Identificador del autor. Ejemplo: `0000150053`.
</ParamField>

## Parámetros de consulta

<PaginationParams />

<ParamField query="sort" type="string" default="citations_desc">
  Ordenamiento con el formato `campo_direccion`. Campos disponibles:
  `citations`, `alphabetical` y `year`. Ejemplo: `year_desc`.
</ParamField>

<ParamField query="view" type="string" default="full">
  Nivel de detalle: `full` o `summary`.
</ParamField>

<Note>
  La estructura de la respuesta es idéntica a la de [Proyectos de una afiliación](/api-reference/affiliation/get-research-projects).
</Note>
