## Parámetros de consulta

<ParamField query="keywords" type="string">
  Texto de búsqueda (título y contenido del producto).
</ParamField>

<PaginationParams />

<ParamField query="sort" type="string" default="citations_desc">
  Ordenamiento con el formato `campo_direccion`, donde la dirección es `asc` o
  `desc`. Campos para productos: `citations`, `alphabetical` (por título) y
  `year` (año de publicación). Ejemplo: `year_desc`.
</ParamField>

<ParamField query="view" type="string" default="full">
  Nivel de detalle de la respuesta: `full` o `summary`.
</ParamField>

### Filtros

Todos los filtros aceptan varios valores separados por coma. Las opciones válidas de cada uno se consultan en [Filtros disponibles de productos](/api-reference/search/works-filters).

<ParamField query="years" type="string">
  Rango de años de publicación: valor mínimo y valor máximo separados por coma.
  Ejemplo: `2020,2026` (desde 2020 hasta 2026).
</ParamField>

<ParamField query="product_types" type="string">
  Tipos de producto. Usa el `value` devuelto por el endpoint de filtros.
  Ejemplo: `impactu_Artículo de revista`.
</ParamField>

<ParamField query="status" type="string">
  Estado de acceso abierto. Valores: `open`, `diamond`, `green`, `hybrid`,
  `gold`, `bronze` y `closed`. Ejemplo: `diamond,gold`.
</ParamField>

<ParamField query="topics" type="string">
  Tópicos. Usa el `value` devuelto por el endpoint de filtros (URL de OpenAlex).
  Ejemplo: `https://openalex.org/T14483`.
</ParamField>

<ParamField query="countries" type="string">
  Países de las afiliaciones de los autores, por código. Ejemplo: `CO`.
</ParamField>

<ParamField query="authors_count" type="string">
  Cantidad de autores. Formatos válidos: `5` (exactamente 5 autores), `5-` (5 o
  más), `-5` (hasta 5) y `5-8` (de 5 a 8). Los valores deben ser mayores que 0 y
  el mínimo no puede ser mayor que el máximo; en otro caso la API responde con
  error 400.
</ParamField>

<ParamField query="authors_ranking" type="string">
  Clasificación de los autores. Ejemplo: `Investigador Junior`.
</ParamField>

<ParamField query="groups_ranking" type="string">
  Clasificación de los grupos de investigación. Ejemplo: `A1,A`.
</ParamField>

<ParamField query="institutions_type" type="string">
  Tipo de institución de las afiliaciones de los autores (según ROR):
  `education`, `funder`, `healthcare`, `company`, `archive`, `nonprofit`,
  `government`, `facility` u `other`. Ejemplo: `education,company`.
</ParamField>

<Note>
  Si no envías `max`, `page` ni `sort`, se aplican `max=10`, `page=1` y
  `sort=citations_desc`. Si envías `max` y `page` sin `sort`, se aplica
  `sort=citations_desc`.
</Note>

## Respuesta

<ResponseField name="data" type="array">
  Lista de productos.
  <Expandable title="propiedades de cada producto">
    <ResponseField name="id" type="string">
      Identificador del producto.
    </ResponseField>
    <ResponseField name="title" type="string">
      Título principal.
    </ResponseField>
    <ResponseField name="titles" type="object[]">
      Títulos por fuente (`title`, `lang`, `source`, `provenance`).
    </ResponseField>
    <ResponseField name="subtitle" type="string | null">
      Subtítulo.
    </ResponseField>
    <ResponseField name="abstracts" type="object[]">
      Resúmenes por fuente. El texto viene como índice invertido (palabra →
      posiciones).
    </ResponseField>
    <ResponseField name="year_published" type="integer">
      Año de publicación.
    </ResponseField>
    <ResponseField name="date_published" type="integer">
      Fecha de publicación en segundos Unix.
    </ResponseField>
    <ResponseField name="language" type="string">
      Idioma.
    </ResponseField>
    <ResponseField name="doi" type="string | null">
      DOI.
    </ResponseField>
    <ResponseField name="authors" type="object[]">
      Autores del producto.
    </ResponseField>
    <ResponseField name="authors_count" type="integer">
      Cantidad de autores.
    </ResponseField>
    <ResponseField name="citations_count" type="object[]">
      Citas por fuente (`count`, `source`, `provenance`).
    </ResponseField>
    <ResponseField name="citations_by_year" type="object[]">
      Citas por año (`year`, `cited_by_count`).
    </ResponseField>
    <ResponseField name="bibliographic_info" type="object">
      Volumen, número, página inicial y final.
    </ResponseField>
    <ResponseField name="source" type="object">
      Fuente (revista) donde se publicó: nombre, ISSN, editorial, APC, rankings
      y tipos.
    </ResponseField>
    <ResponseField name="open_access" type="object">
      Estado de acceso abierto (`is_open_access`, `open_access_status`, `url`,
      `has_repository_fulltext`).
    </ResponseField>
    <ResponseField name="apc" type="object">
      Información de APC del producto.
    </ResponseField>
    <ResponseField name="product_types" type="object[]">
      Tipos de producto (`name`, `source`).
    </ResponseField>
    <ResponseField name="types" type="object[]">
      Tipos del producto por fuente.
    </ResponseField>
    <ResponseField name="primary_topic" type="object">
      Tópico principal, con dominio, campo y subcampo.
    </ResponseField>
    <ResponseField name="topics" type="object[]">
      Tópicos del producto con su `score`.
    </ResponseField>
    <ResponseField name="subjects" type="object[]">
      Áreas de conocimiento agrupadas por fuente.
    </ResponseField>
    <ResponseField name="keywords" type="string[]">
      Palabras clave.
    </ResponseField>
    <ResponseField name="external_ids" type="object[]">
      Identificadores externos (OpenAlex, DOI, MAG, DSpace, etc.).
    </ResponseField>
    <ResponseField name="external_urls" type="object[]">
      Enlaces externos (acceso abierto, repositorio).
    </ResponseField>
    <ResponseField name="groups" type="array">
      Grupos de investigación asociados.
    </ResponseField>
    <ResponseField name="references" type="array">
      Referencias.
    </ResponseField>
    <ResponseField name="updated" type="object[]">
      Última actualización por fuente (`source`, `time` en segundos Unix).
    </ResponseField>
  </Expandable>
</ResponseField>
