## Respuesta

<ResponseField name="data" type="array">
  Lista de patentes.
  <Expandable title="propiedades de cada patente">
    <ResponseField name="id" type="string">
      Identificador de la patente.
    </ResponseField>
    <ResponseField name="title" type="string">
      Título.
    </ResponseField>
    <ResponseField name="titles" type="object[]">
      Títulos por fuente (`title`, `lang`, `source`, `provenance`).
    </ResponseField>
    <ResponseField name="language" type="string">
      Idioma.
    </ResponseField>
    <ResponseField name="authors" type="object[]">
      Autores (inventores), con sus afiliaciones.
    </ResponseField>
    <ResponseField name="authors_count" type="integer">
      Cantidad de autores.
    </ResponseField>
    <ResponseField name="groups" type="object[]">
      Grupos de investigación asociados (`id`, `name`).
    </ResponseField>
    <ResponseField name="product_types" type="object[]">
      Tipos de producto (`name`, `source`).
    </ResponseField>
    <ResponseField name="types" type="object[]">
      Tipos de la patente por fuente, con su nivel (`type`, `level`, `source`).
    </ResponseField>
    <ResponseField name="ranking" type="object[]">
      Clasificaciones (`rank`, `source`, `date`).
    </ResponseField>
    <ResponseField name="external_ids" type="object[]">
      Identificadores externos (`id`, `source`, `provenance`).
    </ResponseField>
    <ResponseField name="external_urls" type="object[]">
      Enlaces externos.
    </ResponseField>
    <ResponseField name="updated" type="object[]">
      Última actualización por fuente (`source`, `time` en segundos Unix).
    </ResponseField>
  </Expandable>
</ResponseField>
