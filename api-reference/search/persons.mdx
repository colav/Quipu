---
title: "Buscar personas"
api: "GET https://api.dev.impactu.colav.co/search/person"
description: "Búsqueda paginada de personas (investigadores) por palabras clave y filtros."
---

import PaginationParams from "/snippets/pagination-params.mdx";
import MetaResponse from "/snippets/meta-response.mdx";

## Parámetros de consulta

<ParamField query="keywords" type="string">
  Texto de búsqueda (nombre o parte del nombre de la persona).
</ParamField>

<PaginationParams />

<ParamField query="countries" type="string">
  Filtra por país.
</ParamField>

<ParamField query="subjects" type="string">
  Filtra por área de conocimiento.
</ParamField>

<ParamField query="view" type="string" default="full">
  Nivel de detalle de la respuesta: `full` o `summary`.
</ParamField>

<Note>
  Si no envías `max`, `page` ni `sort`, se aplican `max=10`, `page=1` y
  `sort=citations_desc`. Si envías `max` y `page` sin `sort`, se aplica
  `sort=citations_desc`.
</Note>

## Respuesta

<ResponseField name="data" type="array">
  Lista de personas.
  <Expandable title="propiedades de cada persona">
    <ResponseField name="id" type="string">
      Identificador de la persona.
    </ResponseField>
    <ResponseField name="full_name" type="string">
      Nombre completo.
    </ResponseField>
    <ResponseField name="first_names" type="string[]">
      Nombres.
    </ResponseField>
    <ResponseField name="last_names" type="string[]">
      Apellidos.
    </ResponseField>
    <ResponseField name="initials" type="string">
      Iniciales.
    </ResponseField>
    <ResponseField name="aliases" type="string[]">
      Variantes del nombre con las que aparece en las fuentes.
    </ResponseField>
    <ResponseField name="sex" type="string | null">
      Sexo.
    </ResponseField>
    <ResponseField name="age" type="integer | null">
      Edad.
    </ResponseField>
    <ResponseField name="marital_status" type="string | null">
      Estado civil.
    </ResponseField>
    <ResponseField name="birthplace" type="object">
      Ciudad (`city`), departamento (`state`) y país (`country`) de nacimiento.
    </ResponseField>
    <ResponseField name="products_count" type="integer">
      Cantidad de productos.
    </ResponseField>
    <ResponseField name="h_index" type="integer">
      Índice h.
    </ResponseField>
    <ResponseField name="h5_index" type="integer">
      Índice h de los últimos 5 años.
    </ResponseField>
    <ResponseField name="citations_count" type="object[]">
      Citaciones por fuente: `count`, `source`, `provenance`.
    </ResponseField>
    <ResponseField name="external_ids" type="object[]">
      Identificadores externos (`id`, `source`, `provenance`), como ORCID,
      OpenAlex, Scienti o Google Scholar.
    </ResponseField>
    <ResponseField name="affiliations" type="object[]">
      Afiliaciones de la persona (institución, facultad, departamento, grupo).
    </ResponseField>
    <ResponseField name="degrees" type="object[]">
      Formación académica.
    </ResponseField>
    <ResponseField name="ranking" type="object[]">
      Categorías y clasificaciones (contrato, jornada, categoría laboral,
      Minciencias).
    </ResponseField>
    <ResponseField name="subjects" type="object[]">
      Áreas de conocimiento agrupadas por fuente.
    </ResponseField>
    <ResponseField name="keywords" type="string[]">
      Palabras clave.
    </ResponseField>
    <ResponseField name="related_works" type="array">
      Trabajos relacionados.
    </ResponseField>
    <ResponseField name="updated" type="object[]">
      Fecha de última actualización por fuente (`source`, `time` en segundos
      Unix).
    </ResponseField>
  </Expandable>
</ResponseField>
