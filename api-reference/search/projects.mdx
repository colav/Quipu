## Respuesta

<ResponseField name="data" type="array">
  Lista de proyectos.
  <Expandable title="propiedades de cada proyecto">
    <ResponseField name="id" type="string">
      Identificador del proyecto.
    </ResponseField>
    <ResponseField name="title" type="string">
      Título.
    </ResponseField>
    <ResponseField name="titles" type="object[]">
      Títulos por fuente (`title`, `lang`, `source`, `provenance`).
    </ResponseField>
    <ResponseField name="abstract" type="string">
      Resumen.
    </ResponseField>
    <ResponseField name="language" type="string">
      Idioma.
    </ResponseField>
    <ResponseField name="keywords" type="string[] | null">
      Palabras clave.
    </ResponseField>
    <ResponseField name="authors" type="object[]">
      Autores (investigadores), con sus afiliaciones.
    </ResponseField>
    <ResponseField name="authors_count" type="integer">
      Cantidad de autores.
    </ResponseField>
    <ResponseField name="groups" type="object[]">
      Grupos de investigación asociados (`id`, `name`).
    </ResponseField>
    <ResponseField name="date_init" type="integer | null">
      Fecha de inicio en segundos Unix.
    </ResponseField>
    <ResponseField name="date_end" type="integer | null">
      Fecha de finalización en segundos Unix.
    </ResponseField>
    <ResponseField name="year_init" type="integer | null">
      Año de inicio.
    </ResponseField>
    <ResponseField name="year_end" type="integer | null">
      Año de finalización.
    </ResponseField>
    <ResponseField name="product_types" type="object[]">
      Tipos de producto (`name`, `source`).
    </ResponseField>
    <ResponseField name="types" type="object[]">
      Tipos del proyecto por fuente, con su nivel (`type`, `code`, `level`,
      `source`).
    </ResponseField>
    <ResponseField name="ranking" type="object[]">
      Clasificaciones.
    </ResponseField>
    <ResponseField name="external_ids" type="object[]">
      Identificadores externos (`id`, `source`, `provenance`).
    </ResponseField>
    <ResponseField name="external_urls" type="object[]">
      Enlaces externos.
    </ResponseField>
    <ResponseField name="updated" type="object[]">
      Última actualización por fuente (`source`, `time` en segundos Unix).
    </ResponseField>
  </Expandable>
</ResponseField>
