## Parámetros de consulta

<ParamField query="keywords" type="string">
  Texto de búsqueda (nombre de la fuente).
</ParamField>

<PaginationParams />

<ParamField query="sort" type="string" default="citations_desc">
  Ordenamiento con el formato `campo_direccion`, donde la dirección es `asc` o
  `desc`. Campos para fuentes: `products` (cantidad de productos), `citations`
  (citas de OpenAlex) y `alphabetical` (por nombre). Ejemplo: `products_desc`.
</ParamField>

<ParamField query="view" type="string" default="full">
  Nivel de detalle de la respuesta: `full` o `summary`.
</ParamField>

### Filtros

Todos los filtros aceptan varios valores separados por coma.

<ParamField query="source_types" type="string">
  Tipo de fuente. No distingue mayúsculas. Ejemplo: `journal,repository`.
</ParamField>

<ParamField query="scimago_quartiles" type="string">
  Cuartil Scimago vigente (Best Quartile). Valores: `Q1`, `Q2`, `Q3`, `Q4` y `-`
  (sin cuartil). Ejemplo: `Q1,Q2`.
</ParamField>

<ParamField query="status" type="string">
  Ruta de acceso abierto. Valores: `diamond` (abierta, sin APC), `gold`
  (abierta, con APC), `hybrid` (cerrada, con APC), `closed` (cerrada, sin APC) y
  `open` (cualquier ruta abierta: `diamond` o `gold`). Ejemplo: `diamond,gold`.
</ParamField>

<ParamField query="apc_range" type="string">
  Rango de APC en USD. Con un valor, trae las fuentes con APC mayor o igual a
  ese valor. Con dos o más valores, usa el menor y el mayor como límites del
  rango. Ejemplo: `100,2000`.
</ParamField>

<ParamField query="publication_time" type="string">
  Tiempo de publicación en semanas. Con un valor, trae las fuentes con tiempo
  mayor o igual a ese valor. Con dos o más valores, usa el menor y el mayor como
  límites del rango (enteros). Ejemplo: `4,12`.
</ParamField>

<ParamField query="license_type" type="string">
  Tipo de licencia. Ejemplo: `CC BY,CC BY-NC`.
</ParamField>

<ParamField query="topics" type="string">
  Identificadores de tópicos de OpenAlex (URL completa). Ejemplo:
  `https://openalex.org/T10017,https://openalex.org/T14434`.
</ParamField>

<Note>
  Si no envías `max`, `page` ni `sort`, se aplican `max=10`, `page=1` y
  `sort=citations_desc`. Si envías `max` y `page` sin `sort`, se aplica
  `sort=citations_desc`. Los valores inválidos de `apc_range` y
  `publication_time` se ignoran sin devolver error.
</Note>

## Respuesta

<ResponseField name="data" type="array">
  Lista de fuentes.
  <Expandable title="propiedades de cada fuente">
    <ResponseField name="id" type="string">
      Identificador de la fuente.
    </ResponseField>
    <ResponseField name="names" type="object[]">
      Nombres por fuente de información (`name`, `lang`, `source`,
      `provenance`).
    </ResponseField>
    <ResponseField name="abbreviations" type="string[]">
      Siglas.
    </ResponseField>
    <ResponseField name="type" type="string">
      Tipo principal (por ejemplo, `journal`).
    </ResponseField>
    <ResponseField name="types" type="object[]">
      Tipos por fuente de información.
    </ResponseField>
    <ResponseField name="publisher" type="object">
      Editorial (`name`, `country_code`, `id`).
    </ResponseField>
    <ResponseField name="languages" type="string[]">
      Idiomas de publicación.
    </ResponseField>
    <ResponseField name="keywords" type="string[]">
      Palabras clave.
    </ResponseField>
    <ResponseField name="products_count" type="integer">
      Cantidad de productos en la base de datos.
    </ResponseField>
    <ResponseField name="citations_count" type="object[]">
      Citas por fuente de información (`count`, `source`, `provenance`).
    </ResponseField>
    <ResponseField name="global_products_count" type="integer">
      Cantidad total de productos de la fuente a nivel global.
    </ResponseField>
    <ResponseField name="global_citations_count" type="integer">
      Cantidad total de citas de la fuente a nivel global.
    </ResponseField>
    <ResponseField name="scimago_best_quartile" type="string | null">
      Mejor cuartil Scimago vigente.
    </ResponseField>
    <ResponseField name="ranking" type="object[]">
      Historial de clasificaciones (Scimago, Publindex y otras) con su vigencia
      (`from_date`, `to_date`).
    </ResponseField>
    <ResponseField name="open_access_status" type="string | null">
      Ruta de acceso abierto (`diamond`, `gold`, `hybrid`, `closed`).
    </ResponseField>
    <ResponseField name="open_access_start_year" type="integer | null">
      Año desde el que es de acceso abierto.
    </ResponseField>
    <ResponseField name="apc" type="object">
      Cargos por procesamiento de artículos (`charges`, `currency`, `paid`).
    </ResponseField>
    <ResponseField name="licenses" type="object[]">
      Licencias (`type`, `url` y banderas `BY`, `NC`, `ND`, `SA`).
    </ResponseField>
    <ResponseField name="copyright" type="object">
      Derechos de autor (`author_retains`, `url`).
    </ResponseField>
    <ResponseField name="waiver" type="object">
      Exención de APC (`has_waiver`, `url`).
    </ResponseField>
    <ResponseField name="publication_time_weeks" type="integer | null">
      Tiempo de publicación en semanas.
    </ResponseField>
    <ResponseField name="review_processes" type="string[]">
      Procesos de revisión.
    </ResponseField>
    <ResponseField name="plagiarism_detection" type="boolean | null">
      Si la fuente usa detección de plagio.
    </ResponseField>
    <ResponseField name="subjects" type="object[]">
      Áreas de conocimiento agrupadas por fuente de información.
    </ResponseField>
    <ResponseField name="topics" type="object[]">
      Tópicos de OpenAlex (dominio, campo, subcampo).
    </ResponseField>
    <ResponseField name="external_ids" type="object[]">
      Identificadores externos (ISSN, OpenAlex, Scimago, Scienti).
    </ResponseField>
    <ResponseField name="external_urls" type="object[]">
      Enlaces externos (sitio de la revista, alcance, instrucciones para
      autores, licencia).
    </ResponseField>
    <ResponseField name="updated" type="object[]">
      Última actualización por fuente de información (`source`, `time` en
      segundos Unix).
    </ResponseField>
  </Expandable>
</ResponseField>
