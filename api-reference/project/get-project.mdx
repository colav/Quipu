---
title: "Obtener un proyecto"
api: "GET https://api.dev.impactu.colav.co/project/{project_id}"
description: "Devuelve un proyecto de investigación a partir de su identificador."
---

## Parámetros de ruta

<ParamField path="project_id" type="string" required>
  Identificador del proyecto. Ejemplo: `6ab4ac4e5418a51e324cc06c`.
</ParamField>

## Respuesta

<ResponseField name="data" type="object">
  El proyecto solicitado.
  <Expandable title="propiedades">
    <ResponseField name="id" type="string">
      Identificador del proyecto.
    </ResponseField>
    <ResponseField name="title" type="string">
      Título principal.
    </ResponseField>
    <ResponseField name="titles" type="object[]">
      Títulos por fuente (`title`, `lang`, `source`, `provenance`).
    </ResponseField>
    <ResponseField name="language" type="string | null">
      Idioma (código ISO 639-1).
    </ResponseField>
    <ResponseField name="abstract" type="string | null">
      Resumen del proyecto.
    </ResponseField>
    <ResponseField name="keywords" type="array | null">
      Palabras clave.
    </ResponseField>
    <ResponseField name="year_init" type="integer | null">
      Año de inicio del proyecto.
    </ResponseField>
    <ResponseField name="year_end" type="integer | null">
      Año de finalización del proyecto. `null` si está vigente.
    </ResponseField>
    <ResponseField name="date_init" type="integer | null">
      Fecha de inicio en segundos Unix.
    </ResponseField>
    <ResponseField name="date_end" type="integer | null">
      Fecha de finalización en segundos Unix. `null` si está vigente.
    </ResponseField>
    <ResponseField name="product_types" type="object[]">
      Tipos de producto por fuente (`name`, `source`). Por ejemplo:
      `"Proyecto de investigación"` o `"Proyecto de extensión"` según
      Minciencias.
    </ResponseField>
    <ResponseField name="types" type="object[]">
      Tipos por fuente con `type`, `level`, `source` y `provenance`.
    </ResponseField>
    <ResponseField name="ranking" type="object[]">
      Clasificaciones del proyecto (`rank`, `source`, `date`, `level`).
    </ResponseField>
    <ResponseField name="authors" type="object[]">
      Autores con sus afiliaciones e identificadores externos.
    </ResponseField>
    <ResponseField name="authors_count" type="integer">
      Cantidad de autores.
    </ResponseField>
    <ResponseField name="groups" type="object[]">
      Grupos de investigación asociados (`id`, `name`).
    </ResponseField>
    <ResponseField name="external_ids" type="object[]">
      Identificadores externos (`id`, `source`, `provenance`).
    </ResponseField>
    <ResponseField name="external_urls" type="object[]">
      Enlaces externos (`url`, `source`).
    </ResponseField>
    <ResponseField name="updated" type="object[]">
      Última actualización por fuente (`source`, `time` en segundos Unix).
    </ResponseField>
  </Expandable>
</ResponseField>
