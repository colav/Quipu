## Path parameters

<ParamField path="project_id" type="string" required>
  Project identifier. Example: `6ab4ac4e5418a51e324cc06c`.
</ParamField>

## Query parameters

<ViewParam />

## Response

<ResponseField name="data" type="object">
  The requested project.
  <Expandable title="properties">
    <ResponseField name="id" type="string">
      Project identifier.
    </ResponseField>
    <ResponseField name="title" type="string">
      Main title.
    </ResponseField>
    <ResponseField name="titles" type="object[]">
      Titles by source (`title`, `lang`, `source`, `provenance`).
    </ResponseField>
    <ResponseField name="language" type="string | null">
      Language (ISO 639-1 code).
    </ResponseField>
    <ResponseField name="abstract" type="string | null">
      Project abstract.
    </ResponseField>
    <ResponseField name="keywords" type="array | null">
      Keywords.
    </ResponseField>
    <ResponseField name="year_init" type="integer | null">
      Start year of the project.
    </ResponseField>
    <ResponseField name="year_end" type="integer | null">
      End year of the project. `null` if it is ongoing.
    </ResponseField>
    <ResponseField name="date_init" type="integer | null">
      Start date in Unix seconds.
    </ResponseField>
    <ResponseField name="date_end" type="integer | null">
      End date in Unix seconds. `null` if it is ongoing.
    </ResponseField>
    <ResponseField name="product_types" type="object[]">
      Product types by source (`name`, `source`). For example:
      `"Proyecto de investigación"` (research project) or `"Proyecto de extensión"` (outreach project) according to
      Minciencias.
    </ResponseField>
    <ResponseField name="types" type="object[]">
      Types by source with `type`, `level`, `source` and `provenance`.
    </ResponseField>
    <ResponseField name="ranking" type="object[]">
      Project classifications (`rank`, `source`, `date`, `level`).
    </ResponseField>
    <ResponseField name="authors" type="object[]">
      Authors with their affiliations and external identifiers.
    </ResponseField>
    <ResponseField name="authors_count" type="integer">
      Number of authors.
    </ResponseField>
    <ResponseField name="groups" type="object[]">
      Associated research groups (`id`, `name`).
    </ResponseField>
    <ResponseField name="external_ids" type="object[]">
      External identifiers (`id`, `source`, `provenance`).
    </ResponseField>
    <ResponseField name="external_urls" type="object[]">
      External links (`url`, `source`).
    </ResponseField>
    <ResponseField name="updated" type="object[]">
      Last update by source (`source`, `time` in Unix seconds).
    </ResponseField>
  </Expandable>
</ResponseField>
