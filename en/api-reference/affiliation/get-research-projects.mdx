---
title: "Affiliation projects"
api: "GET https://api.dev.impactu.colav.co/affiliation/{affiliation_type}/{affiliation_id}/research/projects"
description: "Paginated list of the research projects associated with an affiliation."
---

import PaginationParams from "/snippets/en/pagination-params.mdx";
import MetaResponse from "/snippets/en/meta-response.mdx";
import ViewParam from "/snippets/en/view-param.mdx";

## Path parameters

<ParamField path="affiliation_type" type="string" required>
  Affiliation type: `institution`, `faculty`, `department` or `group`.
</ParamField>

<ParamField path="affiliation_id" type="string" required>
  Affiliation identifier.
</ParamField>

## Query parameters

<PaginationParams />

<ParamField query="sort" type="string" default="citations_desc">
  Sort with the format `field_direction`, where the direction is `asc` or
  `desc`. Available fields: `citations`, `alphabetical` (by title) and
  `year` (start year). Example: `year_desc`.
</ParamField>

<ViewParam />

<Note>
  If you don't send `max`, `page` or `sort`, `max=10`, `page=1` and
  `sort=citations_desc` apply. If you send `max` and `page` without `sort`,
  `sort=citations_desc` applies.
</Note>

## Response

<ResponseField name="data" type="array">
  List of the affiliation's projects. Fields without data are left out.
  <Expandable title="properties of each project">
    <ResponseField name="id" type="string">
      Project identifier.
    </ResponseField>
    <ResponseField name="title" type="string | null">
      Main title.
    </ResponseField>
    <ResponseField name="titles" type="object[]">
      Titles by source (`title`, `lang`, `source`, `provenance`).
    </ResponseField>
    <ResponseField name="language" type="string | null">
      Language.
    </ResponseField>
    <ResponseField name="abstract" type="string | null">
      Project abstract.
    </ResponseField>
    <ResponseField name="year_init" type="integer | string | null">
      Start year.
    </ResponseField>
    <ResponseField name="year_end" type="integer | string | null">
      End year.
    </ResponseField>
    <ResponseField name="date_init" type="integer | null">
      Start date in Unix seconds.
    </ResponseField>
    <ResponseField name="date_end" type="integer | null">
      End date in Unix seconds.
    </ResponseField>
    <ResponseField name="keywords" type="array">
      Keywords.
    </ResponseField>
    <ResponseField name="authors" type="object[]">
      Authors with their affiliations and classifications.
    </ResponseField>
    <ResponseField name="authors_count" type="integer | null">
      Number of authors.
    </ResponseField>
    <ResponseField name="groups" type="object[]">
      Associated research groups, with citations and classifications.
    </ResponseField>
    <ResponseField name="types" type="object[]">
      Project types by source (`type`, `source`, `provenance`).
    </ResponseField>
    <ResponseField name="product_types" type="object[]">
      Types normalized by ImpactU (`name`, `source`).
    </ResponseField>
    <ResponseField name="ranking" type="object[]">
      Project classifications.
    </ResponseField>
    <ResponseField name="external_ids" type="object[]">
      External identifiers (`id`, `source`, `provenance`).
    </ResponseField>
    <ResponseField name="external_urls" type="object[]">
      External links (`url`, `source`).
    </ResponseField>
    <ResponseField name="updated" type="object[]">
      Last update by source (`source`, `time` in Unix seconds).
    </ResponseField>
  </Expandable>
</ResponseField>
