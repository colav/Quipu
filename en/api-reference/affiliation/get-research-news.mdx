---
title: "Affiliation news"
api: "GET https://api.dev.impactu.colav.co/affiliation/{affiliation_type}/{affiliation_id}/research/news"
description: "Paginated list of the news associated with the authors of an affiliation."
---

import PaginationParams from "/snippets/en/pagination-params.mdx";
import MetaResponse from "/snippets/en/meta-response.mdx";

## Path parameters

<ParamField path="affiliation_type" type="string" required>
  Affiliation type: `institution`, `faculty`, `department` or `group`.
</ParamField>

<ParamField path="affiliation_id" type="string" required>
  Affiliation identifier.
</ParamField>

## Query parameters

<PaginationParams />

<Note>
  This endpoint does not accept the `sort` or `view` parameters, or product
  filters. Only `max` and `page` apply.
</Note>

## Response

<ResponseField name="data" type="array">
  List of the affiliation's news.
  <Expandable title="properties of each news item">
    <ResponseField name="url_id" type="string | null">
      Unique identifier of the news URL.
    </ResponseField>
    <ResponseField name="medium_id" type="string | null">
      Identifier of the media outlet.
    </ResponseField>
    <ResponseField name="url" type="string | null">
      Direct URL of the news item.
    </ResponseField>
    <ResponseField name="url_title" type="string | null">
      Title of the news item.
    </ResponseField>
    <ResponseField name="url_language" type="string | null">
      Language of the news item, in English and upper case (for example, `SPANISH`).
    </ResponseField>
    <ResponseField name="url_date" type="string | null">
      Publication date of the news item.
    </ResponseField>
    <ResponseField name="medium" type="string | null">
      URL of the media outlet's website.
    </ResponseField>
  </Expandable>
</ResponseField>
