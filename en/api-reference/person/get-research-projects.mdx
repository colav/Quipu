---

import PaginationParams from "/snippets/en/pagination-params.mdx";
import MetaResponse from "/snippets/en/meta-response.mdx";
import ViewParam from "/snippets/en/view-param.mdx";

## Path parameters

<ParamField path="person_id" type="string" required>
  Author identifier. Example: `0000150053`.
</ParamField>

## Query parameters

<PaginationParams />

<ParamField query="sort" type="string" default="citations_desc">
  Sort with the format `field_direction`. Available fields:
  `citations`, `alphabetical` and `year`. Example: `year_desc`.
</ParamField>

<ViewParam />

<Note>
  The response structure is identical to that of [Affiliation projects](/en/api-reference/affiliation/get-research-projects).
</Note>
