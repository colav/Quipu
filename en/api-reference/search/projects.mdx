## Query parameters

<ParamField query="keywords" type="string">
  Search text (project title).
</ParamField>

<SortAlphabetical />

<ViewParam />

<Note>This endpoint has no filters.</Note>

## Response

<ResponseField name="data" type="array">
  List of projects.
  <Expandable title="properties of each project">
    <ResponseField name="id" type="string">
      Project identifier.
    </ResponseField>
    <ResponseField name="title" type="string">
      Title.
    </ResponseField>
    <ResponseField name="titles" type="object[]">
      Titles by source (`title`, `lang`, `source`, `provenance`).
    </ResponseField>
    <ResponseField name="abstract" type="string">
      Abstract.
    </ResponseField>
    <ResponseField name="language" type="string">
      Language.
    </ResponseField>
    <ResponseField name="keywords" type="string[] | null">
      Keywords.
    </ResponseField>
    <ResponseField name="authors" type="object[]">
      Authors (researchers), with their affiliations.
    </ResponseField>
    <ResponseField name="authors_count" type="integer">
      Number of authors.
    </ResponseField>
    <ResponseField name="groups" type="object[]">
      Associated research groups (`id`, `name`).
    </ResponseField>
    <ResponseField name="date_init" type="integer | null">
      Start date in Unix seconds.
    </ResponseField>
    <ResponseField name="date_end" type="integer | null">
      End date in Unix seconds.
    </ResponseField>
    <ResponseField name="year_init" type="integer | null">
      Start year.
    </ResponseField>
    <ResponseField name="year_end" type="integer | null">
      End year.
    </ResponseField>
    <ResponseField name="product_types" type="object[]">
      Product types (`name`, `source`).
    </ResponseField>
    <ResponseField name="types" type="object[]">
      Project types by source, with their level (`type`, `code`, `level`,
      `source`).
    </ResponseField>
    <ResponseField name="ranking" type="object[]">
      Classifications.
    </ResponseField>
    <ResponseField name="external_ids" type="object[]">
      External identifiers (`id`, `source`, `provenance`).
    </ResponseField>
    <ResponseField name="external_urls" type="object[]">
      External links.
    </ResponseField>
    <ResponseField name="updated" type="object[]">
      Last update by source (`source`, `time` in Unix seconds).
    </ResponseField>
  </Expandable>
</ResponseField>
