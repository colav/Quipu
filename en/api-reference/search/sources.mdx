## Query parameters

<ParamField query="keywords" type="string">
  Search text (source name).
</ParamField>

<PaginationParams />

<ParamField query="sort" type="string" default="citations_desc">
  Sort with the format `field_direction`, where the direction is `asc` or
  `desc`. Fields for sources: `products` (number of products), `citations`
  (OpenAlex citations) and `alphabetical` (by name). Example: `products_desc`.
</ParamField>

<ViewParam />

### Filters

All the filters accept several values separated by commas.

<ParamField query="source_types" type="string">
  Source type. Case-insensitive. Example: `journal,repository`.
</ParamField>

<ParamField query="scimago_quartiles" type="string">
  Current Scimago quartile (Best Quartile). Values: `Q1`, `Q2`, `Q3`, `Q4` and `-`
  (no quartile). Example: `Q1,Q2`.
</ParamField>

<ParamField query="status" type="string">
  Open access route. Values: `diamond` (open, without APC), `gold`
  (open, with APC), `hybrid` (closed, with APC), `closed` (closed, without APC) and
  `open` (any open route: `diamond` or `gold`). Example: `diamond,gold`.
</ParamField>

<ParamField query="apc_range" type="string">
  APC range in USD. With one value, it returns the sources with an APC greater than or equal to
  that value. With two or more values, it uses the smallest and the largest as the limits of the
  range. Example: `100,2000`.
</ParamField>

<ParamField query="publication_time" type="string">
  Publication time in weeks. With one value, it returns the sources with a time
  greater than or equal to that value. With two or more values, it uses the smallest and the largest as the
  limits of the range (integers). Example: `4,12`.
</ParamField>

<ParamField query="license_type" type="string">
  License type. Example: `CC BY,CC BY-NC`.
</ParamField>

<ParamField query="topics" type="string">
  OpenAlex topic identifiers (full URL). Example:
  `https://openalex.org/T10017,https://openalex.org/T14434`.
</ParamField>

<Note>
  If you don't send `max`, `page` or `sort`, `max=10`, `page=1` and
  `sort=citations_desc` apply. If you send `max` and `page` without `sort`,
  `sort=citations_desc`. Invalid values of `apc_range` and
  `publication_time` are ignored without returning an error.
</Note>

## Response

<ResponseField name="data" type="array">
  List of sources.
  <Expandable title="properties of each source">
    <ResponseField name="id" type="string">
      Source identifier.
    </ResponseField>
    <ResponseField name="names" type="object[]">
      Names by information source (`name`, `lang`, `source`,
      `provenance`).
    </ResponseField>
    <ResponseField name="abbreviations" type="string[]">
      Acronyms.
    </ResponseField>
    <ResponseField name="type" type="string">
      Main type (for example, `journal`).
    </ResponseField>
    <ResponseField name="types" type="object[]">
      Types by information source.
    </ResponseField>
    <ResponseField name="publisher" type="object">
      Publisher (`name`, `country_code`, `id`).
    </ResponseField>
    <ResponseField name="languages" type="string[]">
      Publication languages.
    </ResponseField>
    <ResponseField name="keywords" type="string[]">
      Keywords.
    </ResponseField>
    <ResponseField name="products_count" type="integer">
      Number of products in the database.
    </ResponseField>
    <ResponseField name="citations_count" type="object[]">
      Citations by information source (`count`, `source`, `provenance`).
    </ResponseField>
    <ResponseField name="global_products_count" type="integer">
      Total number of products of the source worldwide.
    </ResponseField>
    <ResponseField name="global_citations_count" type="integer">
      Total number of citations of the source worldwide.
    </ResponseField>
    <ResponseField name="scimago_best_quartile" type="string | null">
      Best current Scimago quartile.
    </ResponseField>
    <ResponseField name="ranking" type="object[]">
      History of classifications (Scimago, Publindex and others) with their validity
      (`from_date`, `to_date`).
    </ResponseField>
    <ResponseField name="open_access_status" type="string | null">
      Open access route (`diamond`, `gold`, `hybrid`, `closed`).
    </ResponseField>
    <ResponseField name="open_access_start_year" type="integer | null">
      Year since which it is open access.
    </ResponseField>
    <ResponseField name="apc" type="object">
      Article processing charges (`charges`, `currency`, `paid`).
    </ResponseField>
    <ResponseField name="licenses" type="object[]">
      Licenses (`type`, `url` and the flags `BY`, `NC`, `ND`, `SA`).
    </ResponseField>
    <ResponseField name="copyright" type="object">
      Copyright (`author_retains`, `url`).
    </ResponseField>
    <ResponseField name="waiver" type="object">
      APC waiver (`has_waiver`, `url`).
    </ResponseField>
    <ResponseField name="publication_time_weeks" type="integer | null">
      Publication time in weeks.
    </ResponseField>
    <ResponseField name="review_processes" type="string[]">
      Review processes.
    </ResponseField>
    <ResponseField name="plagiarism_detection" type="boolean | null">
      Whether the source uses plagiarism detection.
    </ResponseField>
    <ResponseField name="subjects" type="object[]">
      Subject areas grouped by information source.
    </ResponseField>
    <ResponseField name="topics" type="object[]">
      OpenAlex topics (domain, field, subfield).
    </ResponseField>
    <ResponseField name="external_ids" type="object[]">
      External identifiers (ISSN, OpenAlex, Scimago, Scienti).
    </ResponseField>
    <ResponseField name="external_urls" type="object[]">
      External links (journal site, scope, instructions for
      authors, license).
    </ResponseField>
    <ResponseField name="updated" type="object[]">
      Last update by information source (`source`, `time` in
      Unix seconds).
    </ResponseField>
    <ResponseField name="addresses" type="array">
      Addresses. Empty for sources.
    </ResponseField>
    <ResponseField name="relations" type="array">
      Relations with other entities. Empty for sources.
    </ResponseField>
    <ResponseField name="affiliation_names" type="null">
      Field not used for sources; it comes as `null`.
    </ResponseField>
    <ResponseField name="date_published" type="null">
      Field not used for sources; it comes as `null`.
    </ResponseField>
    <ResponseField name="review_process" type="null">
      Unused field; the review processes are in `review_processes`.
    </ResponseField>
  </Expandable>
</ResponseField>
