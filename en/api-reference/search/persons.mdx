---
title: "Search persons"
api: "GET https://api.dev.impactu.colav.co/search/person"
description: "Paginated search of persons (researchers) by keywords and filters."
---

import PaginationParams from "/snippets/en/pagination-params.mdx";
import MetaResponse from "/snippets/en/meta-response.mdx";
import ViewParam from "/snippets/en/view-param.mdx";

## Query parameters

<ParamField query="keywords" type="string">
  Search text (name or part of the person's name).
</ParamField>

<PaginationParams />

<ParamField query="sort" type="string" default="citations_desc">
  Sort with the format `field_direction`, where the direction is `asc` or
  `desc`. Fields for persons: `citations`, `products`, `h` and `h5`. Example:
  `products_desc`.
</ParamField>

<ViewParam />

<Note>
  If you don't send `max`, `page` or `sort`, `max=10`, `page=1` and
  `sort=citations_desc` apply. If you send `max` and `page` without `sort`,
  `sort=citations_desc` applies.
</Note>

## Response

<ResponseField name="data" type="array">
  List of persons.
  <Expandable title="properties of each person">
    <ResponseField name="id" type="string">
      Person identifier.
    </ResponseField>
    <ResponseField name="full_name" type="string">
      Full name.
    </ResponseField>
    <ResponseField name="first_names" type="string[]">
      First names.
    </ResponseField>
    <ResponseField name="last_names" type="string[]">
      Last names.
    </ResponseField>
    <ResponseField name="initials" type="string">
      Initials.
    </ResponseField>
    <ResponseField name="aliases" type="string[]">
      Name variants with which the author appears in the sources.
    </ResponseField>
    <ResponseField name="sex" type="string | null">
      Sex.
    </ResponseField>
    <ResponseField name="products_count" type="integer">
      Number of products.
    </ResponseField>
    <ResponseField name="h_index" type="integer">
      H-index.
    </ResponseField>
    <ResponseField name="h5_index" type="integer">
      H-index of the last 5 years.
    </ResponseField>
    <ResponseField name="citations_count" type="object[]">
      Citations by source: `count`, `source`, `provenance`.
    </ResponseField>
    <ResponseField name="external_ids" type="object[]">
      External identifiers (`id`, `source`, `provenance`), such as ORCID,
      OpenAlex, Scienti or Google Scholar.
    </ResponseField>
    <ResponseField name="affiliations" type="object[]">
      Affiliations of the person (institution, faculty, department, group).
      Dates (`start_date`, `end_date`) are in Unix seconds. An
      `end_date` of `-1` means a current affiliation.
    </ResponseField>
    <ResponseField name="degrees" type="object[]">
      Academic degrees.
    </ResponseField>
    <ResponseField name="ranking" type="object[]">
      Categories and classifications (contract, working hours, job category,
      Minciencias).
    </ResponseField>
    <ResponseField name="subjects" type="object[]">
      Subject areas grouped by source.
    </ResponseField>
    <ResponseField name="keywords" type="string[]">
      Keywords.
    </ResponseField>
    <ResponseField name="related_works" type="array">
      Related works.
    </ResponseField>
    <ResponseField name="updated" type="object[]">
      Date of the last update by source (`source`, `time` in Unix
      seconds).
    </ResponseField>
  </Expandable>
</ResponseField>
