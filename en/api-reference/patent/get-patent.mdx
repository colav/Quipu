## Path parameters

<ParamField path="patent_id" type="string" required>
  Patent identifier. Example: `6ab847915418a51e32855aa6`.
</ParamField>

## Query parameters

<ViewParam />

## Response

<ResponseField name="data" type="object">
  The requested patent.
  <Expandable title="properties">
    <ResponseField name="id" type="string">
      Patent identifier.
    </ResponseField>
    <ResponseField name="title" type="string">
      Main title.
    </ResponseField>
    <ResponseField name="titles" type="object[]">
      Titles by source (`title`, `lang`, `source`, `provenance`).
    </ResponseField>
    <ResponseField name="language" type="string | null">
      Language (ISO 639-1 code).
    </ResponseField>
    <ResponseField name="product_types" type="object[]">
      Product types by source (`name`, `source`). For example:
      `"Patente de invención"` according to Minciencias.
    </ResponseField>
    <ResponseField name="types" type="object[]">
      Types by source with `type`, `level`, `source` and `provenance`.
    </ResponseField>
    <ResponseField name="ranking" type="object[]">
      Patent classifications (`rank`, `source`, `date`).
    </ResponseField>
    <ResponseField name="authors" type="object[]">
      Authors with their affiliations and external identifiers.
    </ResponseField>
    <ResponseField name="authors_count" type="integer">
      Number of authors.
    </ResponseField>
    <ResponseField name="groups" type="object[]">
      Associated research groups (`id`, `name`).
    </ResponseField>
    <ResponseField name="external_ids" type="object[]">
      External identifiers (`id`, `source`, `provenance`).
    </ResponseField>
    <ResponseField name="external_urls" type="object[]">
      External links (`url`, `source`).
    </ResponseField>
    <ResponseField name="updated" type="object[]">
      Last update by source (`source`, `time` in Unix seconds).
    </ResponseField>
  </Expandable>
</ResponseField>
